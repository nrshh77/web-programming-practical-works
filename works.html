      Практична робота №6 — SQL

      Приклад SQL-запитів для роботи з таблицею студентів.

      -- Створення таблиці студентів
CREATE TABLE students (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    "group" VARCHAR(50),
    mark INT
);

-- Вибір студентів з оцінкою більше 80
SELECT *
FROM students
WHERE mark > 80
ORDER BY name ASC;

-- Оновлення оцінки студента
UPDATE students
SET mark = 90
WHERE id = 2;

-- Видалення студента
DELETE FROM students
WHERE id = 4;
